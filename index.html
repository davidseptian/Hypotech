// Hypospot Dashboard <-> Google Sheets (dua arah) // Pasang: buka Google Sheet data bulanan > Extensions > Apps Script > tempel kode ini > Deploy > New deployment // Type: Web app | Execute as: Me | Who has access: Anyone > salin URL /exec ke dashboard. // Setelah mengubah kode: Deploy > Manage deployments > Edit > New version. // Pertama kali, jalankan fungsi otorisasi() sekali dari editor untuk memberi izin baca sheet HPP. // Sheet 'Bulanan' = ringkasan per bulan, sheet 'Penjualan' = penjualan per SKU per bulan. const HEAD = ['bulan','revenue','cogs','operating','marketing','net_profit','produk_close','resi_close']; const SH = ['bulan','sku','nama_produk','qty','sub_total']; const HPP_SHEET_ID = '1jGmcijxHcUxReBM2ezBCBL8B8xUKCrTKvsp63vp1NQI'; const json = o => ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); const key = v => v instanceof Date ? Utilities.formatDate(v, SpreadsheetApp.getActiveSpreadsheet().getSpreadsheetTimeZone(), 'yyyy-MM') : String(v).trim(); function otorisasi() { SpreadsheetApp.openById(HPP_SHEET_ID).getName(); } function getSheet(name, head) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sh = ss.getSheetByName(name) || ss.insertSheet(name); if (sh.getLastRow() === 0) sh.appendRow(head); return sh; } function readRows(sh, n) { if (sh.getLastRow() < 2) return []; return sh.getRange(2, 1, sh.getLastRow() - 1, n).getValues().filter(r => r[0] !== '').map(r => { r[0] = key(r[0]); return r; }); } function writeRows(sh, head, rows) { sh.clearContents(); sh.getRange(1, 1, 1, head.length).setValues([head]); if (rows.length) { sh.getRange(2, 1, rows.length, 1).setNumberFormat('@'); sh.getRange(2, 1, rows.length, head.length).setValues(rows); } } // Simpan: bulan yang sudah ada DIGANTI, bulan baru ditambah, bulan di 'del' dihapus (di kedua sheet). function doPost(e) { const lock = LockService.getScriptLock(); lock.waitLock(20000); try { const body = JSON.parse(e.postData.contents); const del = body.del || [], sales = body.sales || {}; const sb = getSheet('Bulanan', HEAD), map = {}; readRows(sb, HEAD.length).forEach(r => { map[r[0]] = r; }); del.forEach(k => delete map[k]); (body.rows || []).forEach(r => { map[r.bulan] = HEAD.map(h => r[h]); }); writeRows(sb, HEAD, Object.keys(map).sort().map(k => map[k])); const sp = getSheet('Penjualan', SH); const touched = {}; del.concat(Object.keys(sales)).forEach(k => { touched[k] = 1; }); const rows = readRows(sp, SH.length).filter(r => !touched[r[0]]); Object.keys(sales).forEach(k => sales[k].forEach(x => rows.push([k].concat(x)))); rows.sort((a, b) => a[0] < b[0] ? -1 : a[0] > b[0] ? 1 : 0); writeRows(sp, SH, rows); return json({ ok: true, rows: Object.keys(map).length }); } finally { lock.releaseLock(); } } // ?action=ping cek koneksi | ?action=data tarik data bulanan + penjualan | ?action=hpp kirim HPP dari tab REKAP SKU function doGet(e) { const act = e.parameter.action; if (act === 'data') { return json({ ok: true, months: readRows(getSheet('Bulanan', HEAD), HEAD.length), sales: readRows(getSheet('Penjualan', SH), SH.length) }); } if (act === 'hpp') { const sh = SpreadsheetApp.openById(HPP_SHEET_ID).getSheetByName('REKAP SKU'); const rows = sh.getDataRange().getValues() .filter(r => /^HS[A-Z]{2,5}\d+$/i.test(String(r[1]).trim())) .map(r => [String(r[1]).trim(), r[4], r[2]]); return json({ ok: true, rows: rows }); } return json({ ok: true }); }
